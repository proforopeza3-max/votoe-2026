Esta URL conectará la página en GitHub con su Google Sheet en la nube.

Pasos para Desplegar el Backend en Google Drive:

  1. Cree una hoja de cálculo nueva en blanco en Google Sheets.
  2. Vaya al menú Extensiones → Apps Script.
  3. Borre el código de ejemplo y pegue el código del script que figura abajo (o desde el archivo google-apps-script/Code.gs).
  4. Haga clic en el botón Implementar → Nueva implementación.
  5. Haga clic en el ícono de engranaje ⚙️ y seleccione Aplicación web.
  6. En "Ejecutar como": seleccione Yo (su correo).
  7. En "Quién tiene acceso": seleccione Cualquier usuario (Anyone). (¡Muy importante!).
  8. Haga clic en Implementar, autorice los permisos y copie la URL de la aplicación web generada.
  9. Pegue la URL en el campo superior de esta pantalla y guarde. ¡Listo! El script creará automáticamente las hojas CONFIG, CANDIDATOS y VOTOS si aún no existen.
/**
 * SISTEMA DE VOTO ELECTRÓNICO ESTUDIANTIL - GOOGLE APPS SCRIPT
 * Péguelo en Extensiones -> Apps Script de su Google Sheet
 */
const SHEET_CONFIG = "CONFIG";
const SHEET_CANDIDATOS = "CANDIDATOS";
const SHEET_VOTOS = "VOTOS";

function doGet(e) {
  try {
    const ss = SpreadsheetApp.getActiveSpreadsheet();
    checkAndInitSheets(ss);
    const action = (e && e.parameter && e.parameter.action) || "getAllData";
    let result = {};

    switch (action) {
      case "ping":
        result = { status: "ok", timestamp: Date.now(), spreadsheetName: ss.getName() };
        break;
      case "getConfig":
        result = { success: true, data: getConfigData(ss) };
        break;
      case "getCandidatos":
        result = { success: true, data: getCandidatosData(ss) };
        break;
      case "getVotos":
      case "getResults":
        result = { success: true, data: getVotosData(ss) };
        break;
      case "getAllData":
      default:
        result = {
          success: true,
          config: getConfigData(ss),
          candidatos: getCandidatosData(ss),
          votos: getVotosData(ss)
        };
        break;
    }
    return ContentService.createTextOutput(JSON.stringify(result)).setMimeType(ContentService.MimeType.JSON);
  } catch (error) {
    return ContentService.createTextOutput(JSON.stringify({ success: false, error: error.toString() })).setMimeType(ContentService.MimeType.JSON);
  }
}

function doPost(e) {
  try {
    const ss = SpreadsheetApp.getActiveSpreadsheet();
    checkAndInitSheets(ss);
    let requestData = {};
    if (e && e.postData && e.postData.contents) {
      try { requestData = JSON.parse(e.postData.contents); } catch (err) { requestData = e.parameter || {}; }
    } else if (e && e.parameter) {
      requestData = e.parameter;
    }

    const action = requestData.action;
    let result = {};

    switch (action) {
      case "vote":
        result = recordVote(ss, requestData);
        break;
      case "saveConfig":
        result = saveConfigData(ss, requestData.config);
        break;
      case "saveCandidatos":
        result = saveCandidatosData(ss, requestData.candidatos);
        break;
      case "resetVotes":
        result = resetVotesData(ss, requestData.adminPass);
        break;
      case "verifyAdmin":
        result = verifyAdminPassword(ss, requestData.password);
        break;
      default:
        result = { success: false, message: "Acción no reconocida" };
        break;
    }
    return ContentService.createTextOutput(JSON.stringify(result)).setMimeType(ContentService.MimeType.JSON);
  } catch (error) {
    return ContentService.createTextOutput(JSON.stringify({ success: false, error: error.toString() })).setMimeType(ContentService.MimeType.JSON);
  }
}

function checkAndInitSheets(ss) {
  if (!ss) ss = SpreadsheetApp.getActiveSpreadsheet();

  // 1. CONFIG
  let sheetConfig = ss.getSheetByName(SHEET_CONFIG);
  if (!sheetConfig) {
    sheetConfig = ss.insertSheet(SHEET_CONFIG);
    sheetConfig.getRange("A1:B1").setValues([["Clave", "Valor"]]).setFontWeight("bold").setBackground("#E2E8F0");
    const defaultPatterns = JSON.stringify({"1":"03678","2":"67852","3":"85210","4":"21036","5":"0124678","6":"0367852"});
    const configRows = [
      ["adminPass", "admin12345"],
      ["tableCount", 6],
      ["tablePatterns", defaultPatterns],
      ["systemTitle", "Elecciones Estudiantiles 2026 - IE 23014 \"ENRIQUE FRACCHIA\""],
      ["systemInsignia", ""]
    ];
    sheetConfig.getRange(2, 1, configRows.length, 2).setValues(configRows);
  }

  // 2. CANDIDATOS
  let sheetCandidatos = ss.getSheetByName(SHEET_CANDIDATOS);
  if (!sheetCandidatos) {
    sheetCandidatos = ss.insertSheet(SHEET_CANDIDATOS);
    sheetCandidatos.getRange("A1:C1").setValues([["ID", "Nombre", "ImagenBase64"]]).setFontWeight("bold").setBackground("#E2E8F0");
    const defaultCandidatos = [
      ["c386467", "LA VOZ DEL FUTURO", ""],
      ["c372475", "ARBOL DE LA SABIDURIA", ""],
      ["c1", "VOTO BLANCO", ""]
    ];
    sheetCandidatos.getRange(2, 1, defaultCandidatos.length, 3).setValues(defaultCandidatos);
  }

  // 3. VOTOS
  let sheetVotos = ss.getSheetByName(SHEET_VOTOS);
  if (!sheetVotos) {
    sheetVotos = ss.insertSheet(SHEET_VOTOS);
    sheetVotos.getRange("A1:D1").setValues([["Timestamp", "Mesa", "CandidatoID", "Hash"]]).setFontWeight("bold").setBackground("#E2E8F0");
  }
}

function getConfigData(ss) {
  const sheet = ss.getSheetByName(SHEET_CONFIG);
  if (!sheet) return {};
  const data = sheet.getDataRange().getValues();
  const config = {};
  for (let i = 1; i < data.length; i++) {
    const key = String(data[i][0]).trim();
    const val = data[i][1];
    if (key) {
      if (key === "tableCount") config[key] = parseInt(val, 10) || 6;
      else if (key === "tablePatterns") {
        try { config[key] = typeof val === "string" ? JSON.parse(val) : val; } catch(e) { config[key] = {}; }
      } else config[key] = val;
    }
  }
  return config;
}

function saveConfigData(ss, newConfig) {
  const sheet = ss.getSheetByName(SHEET_CONFIG);
  if (!sheet || !newConfig) return { success: false };
  const data = sheet.getDataRange().getValues();
  const keyMap = {};
  for (let i = 1; i < data.length; i++) keyMap[String(data[i][0]).trim()] = i + 1;
  for (let key in newConfig) {
    let val = newConfig[key];
    if (key === "tablePatterns" && typeof val !== "string") val = JSON.stringify(val);
    if (keyMap[key]) sheet.getRange(keyMap[key], 2).setValue(val);
    else sheet.appendRow([key, val]);
  }
  return { success: true };
}

function getCandidatosData(ss) {
  const sheet = ss.getSheetByName(SHEET_CANDIDATOS);
  if (!sheet) return [];
  const data = sheet.getDataRange().getValues();
  const c = [];
  for (let i = 1; i < data.length; i++) {
    const id = String(data[i][0]).trim();
    const nombre = String(data[i][1]).trim();
    if (id && nombre) c.push({ id: id, nombre: nombre, imagenBase64: data[i][2] || "" });
  }
  return c;
}

function saveCandidatosData(ss, candidatos) {
  const sheet = ss.getSheetByName(SHEET_CANDIDATOS);
  if (!sheet || !Array.isArray(candidatos)) return { success: false };
  const lastRow = sheet.getLastRow();
  if (lastRow > 1) sheet.getRange(2, 1, lastRow - 1, 3).clearContent();
  if (candidatos.length > 0) {
    const rows = candidatos.map(c => [c.id, c.nombre, c.imagenBase64 || ""]);
    sheet.getRange(2, 1, rows.length, 3).setValues(rows);
  }
  return { success: true };
}

function getVotosData(ss) {
  const sheet = ss.getSheetByName(SHEET_VOTOS);
  if (!sheet) return [];
  const lastRow = sheet.getLastRow();
  if (lastRow <= 1) return [];
  const data = sheet.getRange(2, 1, lastRow - 1, 4).getValues();
  return data.map(d => ({ timestamp: d[0], mesa: d[1], candidatoId: String(d[2]).trim(), hash: d[3] }));
}

function recordVote(ss, payload) {
  const sheet = ss.getSheetByName(SHEET_VOTOS);
  if (!sheet) return { success: false };
  const timestamp = Date.now();
  const rawStr = payload.mesa + "-" + payload.candidatoId + "-" + timestamp;
  const hash = Utilities.base64Encode(rawStr).toUpperCase().substring(0, 16);
  sheet.appendRow([timestamp, payload.mesa, payload.candidatoId, hash]);
  return { success: true, timestamp: timestamp, hash: hash };
}

function resetVotesData(ss, adminPass) {
  const config = getConfigData(ss);
  if (adminPass !== config.adminPass) return { success: false, message: "Clave incorrecta" };
  const sheet = ss.getSheetByName(SHEET_VOTOS);
  const lastRow = sheet.getLastRow();
  if (lastRow > 1) sheet.getRange(2, 1, lastRow - 1, 4).clearContent();
  return { success: true };
}

function verifyAdminPassword(ss, password) {
  const config = getConfigData(ss);
  return { success: (password === config.adminPass) };
}
